name: Unzip and Deploy

on:
  push:
    paths:
      - 'wheelsdrive.zip'

jobs:
  unzip:
    runs-on: ubuntu-latest
    steps:
      - name: Checkout repo
        uses: actions/checkout@v4

      - name: Unzip project
        run: |
          unzip -o wheelsdrive.zip
          cp -r wheelsdrive/* .
          rm -rf wheelsdrive wheelsdrive.zip

      - name: Commit extracted files
        run: |
          git config user.name "github-actions"
          git config user.email "actions@github.com"
          git add .
          git commit -m "Extract WheelsDrive project" || echo "Nothing to commit"
          git push
